    "import pandas as pd, numpy as np, matplotlib\n",
    "matplotlib.use('Agg')\n",
    "import matplotlib.pyplot as plt\n",
    "from sklearn.model_selection import train_test_split, GridSearchCV\n",
    "from sklearn.preprocessing import StandardScaler\n",
    "from sklearn.linear_model import LogisticRegression\n",
    "from sklearn.svm import SVC\n",
    "from sklearn.tree import DecisionTreeClassifier\n",
    "from sklearn.neighbors import KNeighborsClassifier\n",
    "from sklearn.metrics import accuracy_score, confusion_matrix, ConfusionMatrixDisplay\n",
    "\n",
    "df = pd.read_csv('spacex_launch_geo.csv')\n",
    "df['BoosterGroup'] = df['Booster Version'].str.extract(r'(F9 v1\\.0|F9 v1\\.1|F9 FT|F9 B4|F9 B5)')\n",
    "\n",
    "features = pd.get_dummies(df[['Payload Mass (kg)','Orbit','Launch Site','BoosterGroup']],\n",
    "                           columns=['Orbit','Launch Site','BoosterGroup'])\n",
    "X = StandardScaler().fit_transform(features.fillna(0))\n",
    "y = df['class']\n",
    "\n",
    "X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=2)\n",
    "\n",
    "models = {\n",
    "    'Logistic Regression': (LogisticRegression(max_iter=1000), {'C':[0.01,0.1,1,10]}),\n",
    "    'SVM': (SVC(), {'C':[0.1,1,10],'kernel':['linear','rbf']}),\n",
    "    'Decision Tree': (DecisionTreeClassifier(random_state=2), {'max_depth':[2,3,4,5]}),\n",
    "    'KNN': (KNeighborsClassifier(), {'n_neighbors':[2,3,4,5]}),\n",
    "}\n",
    "\n",
    "results = {}\n",
    "best_overall = (None, -1, None)\n",
    "for name, (est, grid) in models.items():\n",
    "    gs = GridSearchCV(est, grid, cv=3)\n",
    "    gs.fit(X_train, y_train)\n",
    "    acc = accuracy_score(y_test, gs.predict(X_test))\n",
    "    results[name] = acc\n",
    "    if acc > best_overall[1]:\n",
    "        best_overall = (name, acc, gs.best_estimator_)\n",
    "\n",
    "print(results)\n",
    "print('Best:', best_overall[0], best_overall[1])\n",
    "\n",
    "# Accuracy bar chart\n",
    "plt.figure(figsize=(8,5))\n",
    "names = list(results.keys()); accs = list(results.values())\n",
    "bars = plt.bar(names, accs, color=['#1565c0','#f9a825','#2e7d32','#8e24aa'])\n",
    "plt.ylim(0,1.1); plt.ylabel('Test Accuracy')\n",
    "plt.title('Classification Model Accuracy Comparison')\n",
    "for b,a in zip(bars,accs):\n",
    "    plt.text(b.get_x()+b.get_width()/2, a+0.02, f'{a:.2f}', ha='center', weight='bold')\n",
    "plt.xticks(rotation=15)\n",
    "plt.tight_layout(); plt.savefig('ml1_accuracy.png'); plt.close()\n",
    "\n",
    "# Confusion matrix for best model\n",
    "y_pred = best_overall[2].predict(X_test)\n",
    "cm = confusion_matrix(y_test, y_pred)\n",
    "disp = ConfusionMatrixDisplay(cm, display_labels=['Failure','Success'])\n",
    "fig, ax = plt.subplots(figsize=(6,5))\n",
    "disp.plot(ax=ax, cmap='Blues', colorbar=False)\n",
    "plt.title(f'Confusion Matrix \u2014 {best_overall[0]} (best model)')\n",
    "plt.tight_layout(); plt.savefig('ml2_confusion.png'); plt.close()\n"
