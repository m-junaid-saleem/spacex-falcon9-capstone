    "import pandas as pd, matplotlib\n",
    "matplotlib.use('Agg')\n",
    "import matplotlib.pyplot as plt\n",
    "\n",
    "df = pd.read_csv('spacex_launch_geo.csv')\n",
    "plt.rcParams['figure.dpi'] = 150\n",
    "\n",
    "sites = df.groupby('Launch Site').agg({'Lat':'first','Long':'first','class':['count','mean']})\n",
    "sites.columns = ['Lat','Long','count','success_rate']\n",
    "sites = sites.reset_index()\n",
    "\n",
    "# Map 1: all launch site locations (static lat/long plot standing in for Folium marker map)\n",
    "plt.figure(figsize=(8,6))\n",
    "plt.scatter(sites['Long'], sites['Lat'], s=200, c='#1565c0', edgecolor='white', zorder=3)\n",
    "for _, r in sites.iterrows():\n",
    "    plt.annotate(r['Launch Site'], (r['Long'], r['Lat']), textcoords=\"offset points\", xytext=(8,5), fontsize=9)\n",
    "plt.xlabel('Longitude'); plt.ylabel('Latitude')\n",
    "plt.title('SpaceX Launch Site Locations (Florida & California)')\n",
    "plt.grid(alpha=0.3)\n",
    "plt.tight_layout(); plt.savefig('m1_launch_sites.png'); plt.close()\n",
    "\n",
    "# Map 2: color-labeled outcomes per site\n",
    "plt.figure(figsize=(8,6))\n",
    "colors = df['class'].map({1:'#2e7d32', 0:'#c62828'})\n",
    "plt.scatter(df['Long'], df['Long']*0+df['Lat'], c=colors, s=90, edgecolor='white', zorder=3)\n",
    "plt.scatter(df['Long'], df['Lat'], c=colors, s=90, edgecolor='white', zorder=3)\n",
    "for _, r in sites.iterrows():\n",
    "    plt.annotate(r['Launch Site'], (r['Long'], r['Lat']), textcoords=\"offset points\", xytext=(8,10), fontsize=9, weight='bold')\n",
    "plt.xlabel('Longitude'); plt.ylabel('Latitude')\n",
    "plt.title('Launch Outcomes by Site (green=success, red=failure)')\n",
    "plt.grid(alpha=0.3)\n",
    "plt.tight_layout(); plt.savefig('m2_outcomes_map.png'); plt.close()\n",
    "\n",
    "# Map 3: proximity illustration for KSC LC-39A (coastline distance annotation)\n",
    "plt.figure(figsize=(7,6))\n",
    "ksc = sites[sites['Launch Site']=='KSC LC-39A'].iloc[0]\n",
    "coastline = (28.56367, -80.5678)\n",
    "plt.scatter([ksc['Long']],[ksc['Lat']], s=250, c='#1565c0', label='KSC LC-39A pad', zorder=3)\n",
    "plt.scatter([coastline[1]],[coastline[0]], s=150, c='#00838f', marker='^', label='Nearest coastline pt', zorder=3)\n",
    "plt.plot([ksc['Long'],coastline[1]],[ksc['Lat'],coastline[0]], 'k--', linewidth=1.5)\n",
    "midx, midy = (ksc['Long']+coastline[1])/2, (ksc['Lat']+coastline[0])/2\n",
    "plt.annotate('~1.3 km', (midx, midy), fontsize=10, weight='bold')\n",
    "plt.legend(); plt.grid(alpha=0.3)\n",
    "plt.xlabel('Longitude'); plt.ylabel('Latitude')\n",
    "plt.title('KSC LC-39A Proximity to Coastline')\n",
    "plt.tight_layout(); plt.savefig('m3_proximity.png'); plt.close()\n",
    "\n",
    "# Dashboard 1: pie chart of success counts by site\n",
    "plt.figure(figsize=(7,7))\n",
    "succ = df[df['class']==1]['Launch Site'].value_counts()\n",
    "plt.pie(succ.values, labels=succ.index, autopct='%1.1f%%', colors=['#1565c0','#2e7d32','#f9a825','#c62828'])\n",
    "plt.title('Launch Success Count by Site')\n",
    "plt.tight_layout(); plt.savefig('d1_success_pie.png'); plt.close()\n",
    "\n",
    "# Dashboard 2: pie chart for highest success ratio site (KSC LC-39A)\n",
    "plt.figure(figsize=(7,7))\n",
    "best_site = sites.sort_values('success_rate', ascending=False).iloc[0]\n",
    "sub = df[df['Launch Site']==best_site['Launch Site']]['class'].value_counts().rename({1:'Success',0:'Failure'})\n",
    "plt.pie(sub.values, labels=sub.index, autopct='%1.1f%%', colors=['#2e7d32','#c62828'])\n",
    "plt.title(f\"Success Ratio \u2014 {best_site['Launch Site']} (highest at {best_site['success_rate']*100:.0f}%)\")\n",
    "plt.tight_layout(); plt.savefig('d2_best_site_pie.png'); plt.close()\n",
    "\n",

    "plt.title('Payload vs. Launch Outcome by Booster Version')\n",
    "plt.legend()\n",
    "plt.tight_layout(); plt.savefig('d3_payload_outcome.png'); plt.close()\n",
